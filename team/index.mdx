export const parseDecisions = (md) =>
  md.split('\n')
    .filter((line) => /^\|\s*\d{4}-\d{2}-\d{2}/.test(line))
    .map((line) => {
      const [date, decision, who, why, task] = line.split('|').slice(1, -1).map((s) => s.trim())
      return { date, decision, who, why, task }
    })

export const DAY = 86400000
export const toDate = (iso) => new Date(`${iso}T12:00:00`)

export const inkOn = (hex = '#999999') => {
  const [r, g, b] = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255)
  return 0.2126 * r + 0.7152 * g + 0.0722 * b > 0.55 ? '#06213f' : '#ffffff'
}

export const Avatar = ({ person, size = 28 }) => (
  <span className="avatar" style={{ width: size, height: size, fontSize: size * 0.42, background: person?.color ?? '#999', color: inkOn(person?.color) }} title={person?.id}>
    {person?.id?.[0] ?? '?'}
  </span>
)
