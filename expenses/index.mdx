export const settleTitle = (w, n) => (n === 0 ? w.settleNone : n === 1 ? w.settleOne : w.settleMany({ number: w.numbers[n] || n }))

export const formatters = (locale) => {
  const money = new Intl.NumberFormat(locale, { style: 'currency', currency: group.currency })
  const day = new Intl.DateTimeFormat(locale, { weekday: 'short', day: 'numeric', month: 'short', timeZone: 'UTC' })
  const short = new Intl.DateTimeFormat(locale, { weekday: 'short', timeZone: 'UTC' })
  const range = new Intl.DateTimeFormat(locale, { day: 'numeric', month: 'long', timeZone: 'UTC' })
  const date = (iso) => new Date(`${iso}T12:00:00Z`)
  return {
    money: (cents) => money.format(cents / 100),
    whole: (cents) => new Intl.NumberFormat(locale, { style: 'currency', currency: group.currency, maximumFractionDigits: 0 }).format(Math.round(cents / 100)),
    day: (iso) => day.format(date(iso)),
    short: (iso) => short.format(date(iso)),
    num: (iso) => date(iso).getUTCDate(),
    dm: (iso) => new Intl.DateTimeFormat(locale, { day: 'numeric', month: 'short', timeZone: 'UTC' }).format(date(iso)),
    range: (a, b) => `${range.format(date(a))} – ${range.format(date(b))}`,
  }
}

export const catName = (w, c) => w.cats?.[c] ?? c

export const PEOPLE = group.people.map((p) => p.name)

export const today = () => {
  const d = new Date()
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
}
export const newId = (prefix) => `${prefix}-${Date.now().toString(36)}${Math.random().toString(36).slice(2, 6)}`

{/* Send one change to the store and report anything it refused. */}
export const send = (change, onProblem) =>
  Promise.resolve(edit(STORE, change)).then((r) => {
    if (r && r.status && r.status !== 'applied' && r.status !== 'unchanged') onProblem(r.reason || r.status)
  }, (err) => onProblem(String(err?.message ?? err)))

export const Face = ({ name, size = 32, ring }) => (
  <span className="face" style={{ width: size, height: size, background: COLOR[name] ?? '#94a3b8', fontSize: size * 0.42, boxShadow: ring ? `0 0 0 3px ${ring}` : undefined }} title={name}>
    {name.slice(0, 1)}
  </span>
)

export const ExpenseForm = ({ initial, title, w, onSave, onCancel }) => {
  const start = parseSplit(initial?.split ?? 'everyone', PEOPLE) ?? PEOPLE.map((p) => [p, 1])
  const [what, setWhat] = React.useState(initial?.what ?? '')
  const [amount, setAmount] = React.useState(initial?.amount ? (initial.amount / 100).toFixed(2) : '')
  const [date, setDate] = React.useState(initial?.date ?? today())
  const [paidBy, setPaidBy] = React.useState(initial?.paid_by ?? PEOPLE[0])
  const [category, setCategory] = React.useState(initial?.category ?? 'Groceries')
  const [weights, setWeights] = React.useState(Object.fromEntries(start))
  const [problem, setProblem] = React.useState(null)
  const toggle = (name) => setWeights((cur) => {
    const next = { ...cur }
    if (next[name]) delete next[name]
    else next[name] = 1
    return next
  })
  const submit = (ev) => {
    ev.preventDefault()
    const cents = Math.round(Number(String(amount).replace(',', '.')) * 100)
    const split = PEOPLE.filter((p) => weights[p]).map((p) => [p, weights[p]])
    if (!what.trim() || !(cents > 0) || split.length === 0) return setProblem(w.incomplete)
    onSave({ date: date || today(), what: what.trim(), category, paid_by: paidBy, amount: cents, split: formatSplit(split, PEOPLE) })
  }
  return (
    <form className="form" onSubmit={submit}>
      <div className="form-title">{title}</div>
      <label className="f-what"><span>{w.fWhat}</span><input value={what} placeholder={w.whatHint} onChange={(e) => setWhat(e.target.value)} autoFocus /></label>
      <label className="f-amount"><span>{w.fAmount}</span><input inputMode="decimal" value={amount} placeholder="0.00" onChange={(e) => setAmount(e.target.value)} /></label>
      <label className="f-date"><span>{w.fDate}</span><input type="date" value={date} onChange={(e) => setDate(e.target.value)} /></label>
      <label className="f-paid"><span>{w.fPaidBy}</span>
        <select value={paidBy} onChange={(e) => setPaidBy(e.target.value)}>{PEOPLE.map((p) => <option key={p} value={p}>{p}</option>)}</select>
      </label>
      <label className="f-cat"><span>{w.fCategory}</span>
        <select value={category} onChange={(e) => setCategory(e.target.value)}>{group.categories.map((c) => <option key={c.name} value={c.name}>{catName(w, c.name)}</option>)}</select>
      </label>
      <fieldset className="f-split">
        <legend>{w.fSplit}</legend>
        {PEOPLE.map((p) => (
          <button type="button" key={p} className={`chip${weights[p] ? ' on' : ''}`} aria-pressed={Boolean(weights[p])} onClick={() => toggle(p)}>
            <Face name={p} size={18} />{p}{weights[p] > 1 ? ` ×${weights[p]}` : ''}
          </button>
        ))}
      </fieldset>
      {problem && <div className="form-problem">{problem}</div>}
      <div className="form-actions">
        <button type="button" className="btn" onClick={onCancel}>{w.cancel}</button>
        <button type="submit" className="btn primary">{w.save}</button>
      </div>
    </form>
  )
}

export const ByCategory = ({ ledger, w, f }) => {
  const rows = group.categories
    .map((c) => ({ ...c, cents: ledger.expenses.filter((e) => e.category === c.name).reduce((s, e) => s + e.cents, 0) }))
    .filter((c) => c.cents > 0)
    .sort((a, b) => b.cents - a.cents)
  const max = Math.max(...rows.map((r) => r.cents), 1)
  return (
    <section className="panel">
      <h3>{w.byCategory}</h3>
      <div className="bars">
        {rows.map((r) => (
          <div className="bar-row" key={r.name}>
            <span className="bar-label"><i style={{ background: r.color }} />{catName(w, r.name)}</span>
            <span className="bar-track"><span className="bar-fill" style={{ width: `${(r.cents / max) * 100}%`, background: r.color }} /></span>
            <span className="bar-value">{f.money(r.cents)}<small>{Math.round((r.cents / Math.max(ledger.total, 1)) * 100)}%</small></span>
          </div>
        ))}
      </div>
    </section>
  )
}

export const ByDay = ({ ledger, span, w, f }) => {
  const days = []
  for (let d = new Date(`${span.from}T12:00:00Z`); d <= new Date(`${span.to}T12:00:00Z`); d.setUTCDate(d.getUTCDate() + 1)) days.push(d.toISOString().slice(0, 10))
  // A short stay shows each day; a month shows weeks, counted from the first day.
  const weekly = days.length > 10
  const buckets = weekly
    ? days.filter((_, i) => i % 7 === 0).map((iso, k) => ({ iso, set: new Set(days.slice(k * 7, k * 7 + 7)) }))
    : days.map((iso) => ({ iso, set: new Set([iso]) }))
  const totals = buckets.map((b) => ({ iso: b.iso, cents: ledger.expenses.filter((e) => b.set.has(e.date)).reduce((s, e) => s + e.cents, 0) }))
  const max = Math.max(...totals.map((t) => t.cents), 1)
  const [hover, setHover] = React.useState(null)
  return (
    <section className="panel">
      <h3>{weekly ? w.byWeek : w.byDay}</h3>
      <div className="columns" onMouseLeave={() => setHover(null)}>
        {totals.map((t) => (
          <div className={`col${hover === t.iso ? ' on' : ''}`} key={t.iso} onMouseEnter={() => setHover(t.iso)}>
            <span className="col-value">{f.whole(t.cents)}</span>
            <span className="col-track"><span className="col-fill" style={{ height: `${Math.max(2, (t.cents / max) * 100)}%` }} /></span>
            <span className="col-day">{weekly ? w.weekOf : f.short(t.iso)}<b>{weekly ? f.dm(t.iso) : f.num(t.iso)}</b></span>
          </div>
        ))}
      </div>
    </section>
  )
}

export const List = ({ ledger, rows: stored, w, f, canEdit, onSave, onDelete }) => {
  const [who, setWho] = React.useState(null)
  const [cat, setCat] = React.useState(null)
  const [editing, setEditing] = React.useState(null)
  const [confirming, setConfirming] = React.useState(null)
  const cats = group.categories.filter((c) => ledger.expenses.some((e) => e.category === c.name))
  const rows = ledger.expenses.filter((e) => (!who || e.shares[who] !== undefined || e.paidBy === who) && (!cat || e.category === cat))
  const shownTotal = rows.reduce((s, e) => s + (who ? e.shares[who] ?? 0 : e.cents), 0)
  const byDay = rows.reduce((acc, e) => ((acc[e.date] ??= []).push(e), acc), {})
  return (
    <section className="list">
      <div className="list-head">
        <h3>{w.list}</h3>
        <span className="list-count">{w.shown({ count: rows.length, total: who ? `${who} · ${w.theirShare} ${f.money(shownTotal)}` : f.money(shownTotal) })}</span>
      </div>
      <div className="filters">
        <button className={`chip${!who ? ' on' : ''}`} onClick={() => setWho(null)} aria-pressed={!who}>{w.everyone}</button>
        {group.people.map(({ name }) => (
          <button key={name} className={`chip${who === name ? ' on' : ''}`} onClick={() => setWho(who === name ? null : name)} aria-pressed={who === name}>
            <Face name={name} size={18} />{name}
          </button>
        ))}
        <span className="sep" />
        <button className={`chip${!cat ? ' on' : ''}`} onClick={() => setCat(null)} aria-pressed={!cat}>{w.all}</button>
        {cats.map((c) => (
          <button key={c.name} className={`chip${cat === c.name ? ' on' : ''}`} onClick={() => setCat(cat === c.name ? null : c.name)} aria-pressed={cat === c.name}>
            <i className="dot" style={{ background: c.color }} />{catName(w, c.name)}
          </button>
        ))}
      </div>
      {rows.length === 0 && <div className="empty">{w.none}</div>}
      <div className="days">
      {Object.entries(byDay).map(([iso, items]) => (
        <div className="day" key={iso}>
          <div className="day-head">
            <span>{f.day(iso)}</span>
            <span>{f.money(items.reduce((s, e) => s + (who ? e.shares[who] ?? 0 : e.cents), 0))}</span>
          </div>
          {items.map((e) => {
            if (canEdit && editing === e.id && stored?.[e.id]) {
              return (
                <div className="row-form" key={e.id}>
                  <ExpenseForm initial={stored[e.id]} title={w.editTitle} w={w} onCancel={() => setEditing(null)} onSave={(values) => { setEditing(null); onSave(e.id, values) }} />
                </div>
              )
            }
            const names = Object.keys(e.shares)
            const all = names.length === group.people.length && names.every((n) => e.weights[n] === 1)
            return (
              <div className="row" key={e.id}>
                <i className="row-cat" style={{ background: CAT[e.category] ?? '#94a3b8' }} />
                <div className="row-main">
                  <div className="row-what">{e.what}</div>
                  <div className="row-meta">
                    <span>{catName(w, e.category)}</span>
                    <span className="row-pay"><Face name={e.paidBy} size={16} />{w.paidBy} {e.paidBy}</span>
                    {!all && (
                      <span className="row-split">
                        {w.splitWith}
                        {names.map((n) => <span key={n} className="mini"><Face name={n} size={16} />{e.weights[n] > 1 ? `×${e.weights[n]}` : ''}</span>)}
                      </span>
                    )}
                  </div>
                </div>
                <div className="row-amount">
                  <b>{f.money(who ? e.shares[who] ?? 0 : e.cents)}</b>
                  {who && <small>{`${w.theirShare} · ${f.money(e.cents)}`}</small>}
                  {canEdit && (confirming === e.id ? (
                    <span className="row-actions">
                      <span className="sure">{w.sure}</span>
                      <button className="link danger" onClick={() => { setConfirming(null); onDelete(e.id) }}>{w.del}</button>
                      <button className="link" onClick={() => setConfirming(null)}>{w.keep}</button>
                    </span>
                  ) : (
                    <span className="row-actions">
                      <button className="link" onClick={() => { setConfirming(null); setEditing(e.id) }}>{w.edit}</button>
                      <button className="link" onClick={() => setConfirming(e.id)}>{w.del}</button>
                    </span>
                  ))}
                </div>
              </div>
            )
          })}
        </div>
      ))}
      </div>
    </section>
  )
}
