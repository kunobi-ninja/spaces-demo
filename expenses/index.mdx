export const lang = () => {

  return WORDS[tag] ? tag : 'en'

export const formatters = (code) => {
  const locale = code === 'es' ? 'es-ES' : 'en-GB'
  const money = new Intl.NumberFormat(locale, { style: 'currency', currency: group.currency })
  const day = new Intl.DateTimeFormat(locale, { weekday: 'short', day: 'numeric', month: 'short', timeZone: 'UTC' })
  const short = new Intl.DateTimeFormat(locale, { weekday: 'short', timeZone: 'UTC' })
  const range = new Intl.DateTimeFormat(locale, { day: 'numeric', month: 'long', timeZone: 'UTC' })
  const date = (iso) => new Date(`${iso}T12:00:00Z`)
  return {
    money: (cents) => money.format(cents / 100),
    whole: (cents) => new Intl.NumberFormat(locale, { style: 'currency', currency: group.currency, maximumFractionDigits: 0 }).format(Math.round(cents / 100)),
    original: (amount, currency) => new Intl.NumberFormat(locale, { style: 'currency', currency }).format(amount),
    day: (iso) => day.format(date(iso)),
    short: (iso) => short.format(date(iso)),
    num: (iso) => date(iso).getUTCDate(),
    dm: (iso) => new Intl.DateTimeFormat(locale, { day: 'numeric', month: 'short', timeZone: 'UTC' }).format(date(iso)),
    range: (a, b) => `${range.format(date(a))} – ${range.format(date(b))}`,
  }
}

export const Balances = ({ ledger, w, f }) => {
  const max = Math.max(...Object.values(ledger.balance).map(Math.abs), 1)
  return (
    <section className="balances">
      {group.people.map(({ name }) => {
        const b = ledger.balance[name]
        const pct = (Math.abs(b) / max) * 50
        return (
          <div className="balance" key={name}>
            <div className="balance-top">
              <Face name={name} size={36} />
              <div>
                <div className="balance-name">{name}</div>
                <div className="balance-sub">{w.paid} {f.money(ledger.paid[name])} · {w.share} {f.money(ledger.owed[name])}</div>
              </div>
            </div>
            <div className="meter" aria-hidden="true">
              <span className="zero" />
              <span className={`fill ${b >= 0 ? 'up' : 'down'}`} style={b >= 0 ? { left: '50%', width: `${pct}%` } : { right: '50%', width: `${pct}%` }} />
            </div>
            <div className={`balance-value ${b > 0 ? 'up' : b < 0 ? 'down' : ''}`}>
              {b === 0 ? w.even : `${b > 0 ? w.getsBack : w.owes} ${f.money(Math.abs(b))}`}
            </div>
          </div>
        )
      })}
    </section>
  )
}

export const ByCategory = ({ ledger, w, f }) => {
  const rows = group.categories
    .map((c) => ({ ...c, cents: ledger.expenses.filter((e) => e.category === c.name).reduce((s, e) => s + e.cents, 0) }))
    .filter((c) => c.cents > 0)
    .sort((a, b) => b.cents - a.cents)
  const max = Math.max(...rows.map((r) => r.cents), 1)
  return (
    <section className="panel">
      <h3>{w.byCategory}</h3>
      <div className="bars">
        {rows.map((r) => (
          <div className="bar-row" key={r.name}>
            <span className="bar-label"><i style={{ background: r.color }} />{r.name}</span>
            <span className="bar-track"><span className="bar-fill" style={{ width: `${(r.cents / max) * 100}%`, background: r.color }} /></span>
            <span className="bar-value">{f.money(r.cents)}<small>{Math.round((r.cents / ledger.total) * 100)}%</small></span>
          </div>
        ))}
      </div>
    </section>
  )
}

export const ByDay = ({ ledger, w, f }) => {
  const days = []
  for (let d = new Date(`${group.from}T12:00:00Z`); d <= new Date(`${group.to}T12:00:00Z`); d.setUTCDate(d.getUTCDate() + 1)) days.push(d.toISOString().slice(0, 10))
  // A short stay shows each day; a month shows weeks, counted from the first day.
  const weekly = days.length > 10
  const buckets = weekly
    ? days.filter((_, i) => i % 7 === 0).map((iso, k) => ({ iso, set: new Set(days.slice(k * 7, k * 7 + 7)) }))
    : days.map((iso) => ({ iso, set: new Set([iso]) }))
  const totals = buckets.map((b) => ({ iso: b.iso, cents: ledger.expenses.filter((e) => b.set.has(e.date)).reduce((s, e) => s + e.cents, 0) }))
  const max = Math.max(...totals.map((t) => t.cents), 1)
  const [hover, setHover] = React.useState(null)
  return (
    <section className="panel">
      <h3>{weekly ? w.byWeek : w.byDay}</h3>
      <div className="columns" onMouseLeave={() => setHover(null)}>
        {totals.map((t) => (
          <div className={`col${hover === t.iso ? ' on' : ''}`} key={t.iso} onMouseEnter={() => setHover(t.iso)}>
            <span className="col-value">{f.whole(t.cents)}</span>
            <span className="col-track"><span className="col-fill" style={{ height: `${Math.max(2, (t.cents / max) * 100)}%` }} /></span>
            <span className="col-day">{weekly ? w.weekOf : f.short(t.iso)}<b>{weekly ? f.dm(t.iso) : f.num(t.iso)}</b></span>
          </div>
        ))}
      </div>
    </section>
  )
}

export const List = ({ ledger, w, f }) => {
  const [who, setWho] = React.useState(null)
  const [cat, setCat] = React.useState(null)
  const cats = group.categories.filter((c) => ledger.expenses.some((e) => e.category === c.name))
  const rows = ledger.expenses.filter((e) => (!who || e.shares[who] !== undefined || e.paidBy === who) && (!cat || e.category === cat))
  const shownTotal = rows.reduce((s, e) => s + (who ? e.shares[who] ?? 0 : e.cents), 0)
  const byDay = rows.reduce((acc, e) => ((acc[e.date] ??= []).push(e), acc), {})
  return (
    <section className="list">
      <div className="list-head">
        <h3>{w.list}</h3>
        <span className="list-count">{w.shown(rows.length, who ? `${who} · ${w.theirShare} ${f.money(shownTotal)}` : f.money(shownTotal))}</span>
      </div>
      <div className="filters">
        <button className={`chip${!who ? ' on' : ''}`} onClick={() => setWho(null)} aria-pressed={!who}>{w.everyone}</button>
        {group.people.map(({ name }) => (
          <button key={name} className={`chip${who === name ? ' on' : ''}`} onClick={() => setWho(who === name ? null : name)} aria-pressed={who === name}>
            <Face name={name} size={18} />{name}
          </button>
        ))}
        <span className="sep" />
        <button className={`chip${!cat ? ' on' : ''}`} onClick={() => setCat(null)} aria-pressed={!cat}>{w.all}</button>
        {cats.map((c) => (
          <button key={c.name} className={`chip${cat === c.name ? ' on' : ''}`} onClick={() => setCat(cat === c.name ? null : c.name)} aria-pressed={cat === c.name}>
            <i className="dot" style={{ background: c.color }} />{c.name}
          </button>
        ))}
      </div>
      {rows.length === 0 && <div className="empty">{w.none}</div>}
      <div className="days">
      {Object.entries(byDay).map(([iso, items]) => (
        <div className="day" key={iso}>
          <div className="day-head">
            <span>{f.day(iso)}</span>
            <span>{f.money(items.reduce((s, e) => s + (who ? e.shares[who] ?? 0 : e.cents), 0))}</span>
          </div>
          {items.map((e, i) => {
            const names = Object.keys(e.shares)
            const all = names.length === group.people.length && names.every((n) => e.weights[n] === 1)
            return (
              <div className="row" key={`${iso}-${i}`}>
                <i className="row-cat" style={{ background: CAT[e.category] ?? '#94a3b8' }} />
                <div className="row-main">
                  <div className="row-what">{e.what}</div>
                  <div className="row-meta">
                    <span>{e.category}</span>
                    <span className="row-pay"><Face name={e.paidBy} size={16} />{w.paidBy} {e.paidBy}</span>
                    {!all && (
                      <span className="row-split">
                        {w.splitWith}
                        {names.map((n) => <span key={n} className="mini"><Face name={n} size={16} />{e.weights[n] > 1 ? `×${e.weights[n]}` : ''}</span>)}
                      </span>
                    )}
                  </div>
                </div>
                <div className="row-amount">
                  <b>{f.money(who ? e.shares[who] ?? 0 : e.cents)}</b>
                  <small>{who ? `${w.theirShare} · ${f.money(e.cents)}` : e.currency !== group.currency ? f.original(e.amount, e.currency) : ''}</small>
                </div>
              </div>
            )
          })}
        </div>
      ))}
      </div>
    </section>
  )
}
