export const words = () => {
  const tag = (typeof navigator !== 'undefined' && navigator.language) || 'en'
  return WORDS[tag.toLowerCase().split('-')[0]] ?? WORDS.en
}

export const Page = ({ view }) => {
  const w = words()
  const store = useStoreStatus(STORE)
  const canEdit = store?.status === 'open' && Boolean(store?.canEdit)
  return (
    <>
      <ContentCalendar posts={posts} settings={settings} words={w} initial={view} />
      <div className="cc-how">
        <style>{`
          .cc-how { width: min(var(--kunobi-prose-width, 1200px), 100%); margin: 0 auto; box-sizing: border-box; padding: 0 24px 40px; font-family: Figtree, system-ui, sans-serif; color: hsl(var(--muted-foreground)); font-size: 14px; line-height: 1.5; }

          .cc-how q { quotes: none; color: hsl(var(--foreground)); font-weight: 600; }
          .cc-how q::before, .cc-how q::after { content: none; }
        `}</style>
        <div>
          <span>{canEdit ? w.changeEdit : w.change}</span>
          {w.prompts.map((p) => <q key={p}>{p}</q>)}
        </div>
      </div>
    </>
  )
}

<Page view={props.view} />
