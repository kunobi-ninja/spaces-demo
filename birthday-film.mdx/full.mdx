export { createCut } from "./src/index"
import { Opening, Memory, Cake, Finale } from "./src/index"
import song from "./music/song.mdx"
import filmData from "./data/film.json"
import { i18n } from "../lib/i18n"
import translations from "./locales/index"

export const music = { score: song, gain: 0.8 }

{/* English says "40th"; other languages show the plain number. */}

export const ordinal = (n, lang) => {
  if (lang !== "en") return String(n)
  const s = n % 100 >= 11 && n % 100 <= 13 ? "th" : ["th", "st", "nd", "rd"][n % 10] ?? "th"
  return `${n}${s}`
}

{/* The film's own words. Their translations, and those of data/film.json, are in locales/<lang>.json; the film follows the app's language. */}

export const WORDS = { turns: "turns {{age}}", happy: "Happy {{age}}, {{to}}", from: "from {{name}}" }

export const t = i18n(WORDS, translations)
export const w = t.w
export const film = t.data("film", filmData)
export const m = film.memories

A thirty-second birthday film in a paper-cut world: the name pops up over the
hills, three memories unfold as pop-up cards, candles shaped like the age light
on the beat, and a card signs off. Change `data/film.json` (the name, the age,
three memories and who it is from; translated in `locales/`) and the whole
film follows. Icons: tent, bike, pasta, plane, music, heart, star.
