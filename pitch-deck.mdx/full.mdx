export const pickLang = () => {
  const tag = (typeof navigator !== "undefined" && navigator.language) || "en"
  const lang = tag.toLowerCase().split("-")[0]
  return WORDS[lang] ? lang : "en"
}

export const deck = DECKS[pickLang()] ?? DECKS.en
