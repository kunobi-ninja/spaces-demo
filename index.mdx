export const Waves = () => {
  const ref = React.useRef(null)
  React.useEffect(() => {
    const canvas = ref.current
    const ctx = canvas.getContext('2d')
    const still = typeof matchMedia === 'function' && matchMedia('(prefers-reduced-motion: reduce)').matches
    const lines = [
      { color: COLORS.sun, amp: 0.13, freq: 1.3, speed: 0.22, y: 0.62, alpha: 0.55 },
      { color: COLORS.ocean, amp: 0.1, freq: 1.8, speed: -0.17, y: 0.66, alpha: 0.45 },
      { color: COLORS.tide, amp: 0.16, freq: 0.9, speed: 0.12, y: 0.72, alpha: 0.4 },
      { color: COLORS.horizon, amp: 0.08, freq: 2.4, speed: 0.28, y: 0.58, alpha: 0.5 },
    ]
    let w = 0
    let h = 0
    const size = () => {
      const r = canvas.getBoundingClientRect()
      const dpr = Math.min(2, globalThis.devicePixelRatio || 1)
      w = r.width
      h = r.height
      canvas.width = Math.round(w * dpr)
      canvas.height = Math.round(h * dpr)
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
    }
    const draw = (t) => {
      ctx.clearRect(0, 0, w, h)
      ctx.globalCompositeOperation = 'lighter'
      for (const l of lines) {
        for (let k = 0; k < 14; k++) {
          ctx.beginPath()
          const off = k * 0.012
          for (let x = 0; x <= w; x += 6) {
            const u = x / w
            const y = h * (l.y + off) + Math.sin(u * Math.PI * 2 * l.freq + t * l.speed + k * 0.18) * h * l.amp * (0.6 + 0.4 * Math.sin(u * Math.PI))
            x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
          }
          ctx.strokeStyle = l.color
          ctx.globalAlpha = l.alpha * (1 - k / 14) * 0.5
          ctx.lineWidth = 1.2
          ctx.stroke()
        }
      }
      ctx.globalAlpha = 1
      ctx.globalCompositeOperation = 'source-over'
    }
    const start = performance.now() / 1000
    size()
    const watch = new ResizeObserver(() => { size(); draw(still ? 4 : performance.now() / 1000) })
    watch.observe(canvas)
    let raf = 0
    if (still) draw(4)
    else {
      const frame = (now) => { draw(now / 1000); raf = requestAnimationFrame(frame) }
      raf = requestAnimationFrame(frame)
    }
    return () => { cancelAnimationFrame(raf); watch.disconnect() }
  }, [])
  return <canvas ref={ref} className="waves" aria-hidden="true" />
}
