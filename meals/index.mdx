export const Meals = () => {
  const t = useI18n(WORDS, translations)
  // Translated recipes keep the list ids of the recipes as written, so ticks hold in every language.
  const recipes = React.useMemo(() => keepKeys(t.data('recipes', data).recipes, data.recipes), [t])
  const localize = React.useCallback((live) => t.data('shopping', live, startShopping), [t])
  return <MealPlan startWeek={startWeek} startShopping={startShopping} localizeShopping={localize} recipes={recipes} aisles={data.aisles} words={t.w} />
}
