import * as React from 'react'
import '@fontsource/parkinsans/600.css'
import '@fontsource/parkinsans/800.css'
import '@fontsource/figtree/400.css'
import '@fontsource/figtree/600.css'
import { TripPlanner } from './src/trip'
import { useI18n } from '../lib/i18n'
import translations from './locales/index'

{/* The places live in the trip-places store, started from data/places.start.json; data/places.json is its saved copy. The trip's frame and the page's words are below; their translations, and those of the starting places (by id), are in locales/<lang>.json. A place still as it started shows translated. */}

export const TRIP = {
  friends: [
    { name: 'Maya', color: '#f3aa08' },
    { name: 'Leo', color: '#5bcbf9' },
  ],
}

export const WORDS = {
  eyebrow: 'Weekend trip', city: 'Lisbon',
  dates: 'Friday 13 to Sunday 15 November',
  days: ['Fri 13', 'Sat 14', 'Sun 15'],
  all: 'All days', map: 'Map of the trip',
  places: 'places', each: 'each', toBook: 'to book', pp: 'each',
  booked: 'Booked', notBooked: 'Not booked yet', markBooked: 'Mark booked', unbook: 'Not booked', saving: 'Saving…', loading: 'Loading…', unavailable: "This can't be opened right now.", addedBy: 'Added by', close: 'Close', free: 'Free',
  stillToBook: 'Still to book', allBooked: 'Everything is booked.', budget: 'Budget per person',
  kinds: { stay: 'Stay', food: 'Food', see: 'See', coffee: 'Coffee', viewpoint: 'Viewpoint' },
  how: ['The plan is saved in', 'Tap a place to mark it booked, or ask your AI:', 'Ask your AI:'],
  prompts: ['Add the Oceanário on Sunday at 15:00, added by Leo', 'Mark the Jerónimos tickets as booked'],
}

export const Lisbon = () => {
  const t = useI18n(WORDS, translations)
  const w = t.w
  const trip = { ...TRIP, city: w.city, days: w.days.map((label, i) => ({ day: i + 1, label })) }
  const localize = React.useCallback((live) => t.data('places', live), [t])
  return <TripPlanner localize={localize} trip={trip} words={w} />
}

<Lisbon />
