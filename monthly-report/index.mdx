export const ptsText = (v) => `${v > 0 ? '+' : v < 0 ? '−' : '±'}${Math.abs(v * 100).toFixed(1)}`

{/* The three summary lines, put together from the sentences in WORDS.sum. */}

export const summaries = (s, lang) => {
  const dir = (v, flat, up, down) => (Math.abs(v) < 0.005 ? flat : (v > 0 ? up : down)({ pct: pctText(v) }))
  return {
    sumRevenue: ({ month, value, mom, yoy, prev, year }) => {
      let text = s.revenue({ month, value })
      if (mom !== null) text += s.revenueMom({ change: dir(mom, s.revenueFlat, s.revenueUp, s.revenueDown), prev })
      if (yoy !== null) text += s.revenueYoy({ change: dir(yoy, s.revenueFlat, s.revenueUp, s.revenueDown), year })
      return text + s.end
    },
    sumDrivers: ({ orders, aov, move, channel }) => {
      const change = (v) => dir(v, s.driversFlat, s.driversUp, s.driversDown)
      let text = s.drivers({ orders: change(orders), aov: change(aov) })
      if (move) text += s.channelMove({ channel, diff: `${move.diff > 0 ? '+' : '−'}${Math.abs(move.diff).toLocaleString(lang)}` })
      return text
    },
    sumCustomers: ({ share, change, top, topValue }) => {
      let text = s.customers({ share: pctText(share) })
      if (change !== null) text += s.customersChange({ points: ptsText(change) })
      return text + s.customersTop({ top, topValue })
    },
  }
}

export const Page = ({ month }) => {
  const t = useI18n(WORDS, translations)
  const w = { ...t.w, ...summaries(t.w.sum, t.lang) }
  return <Report data={t.data('report', data)} w={w} locale={t.tag.toLowerCase().startsWith(t.lang.toLowerCase().split('-')[0]) ? t.tag : t.lang} initial={month} />
}
