import '@fontsource/parkinsans/800.css'
import '@fontsource/figtree/400.css'
import '@fontsource/figtree/600.css'
import { Viewer } from './src/Viewer'
import scene from './data/scene.json'
import part1 from './assets/felipe-1.spz'
import part2 from './assets/felipe-2.spz'
import part3 from './assets/felipe-3.spz'

{/* The scan, in parts under 16 MiB each. To show your own scan, convert it to SPZ v3, put it in assets/, import it here, and adjust data/scene.json. */}

export const MODELS = [part1, part2, part3]

export const WORDS = {
  en: {
    loading: 'Loading the scan…',
    failed: 'The scan could not be loaded.',
    hint: 'Drag to turn · Scroll or pinch to zoom',
    front: 'Front', side: 'Side', close: 'Close-up',
    reveal: 'Reveal', points: 'Scatter', light: 'Light',
    change: 'How to change this',
    changeLine: 'The camera, title and credit live in data/scene.json. Ask your AI:',
    prompts: ['“Add a top view preset to the scan viewer.”', '“Use my scan assets/chair.spz instead, framed from the front.”'],
  },
  es: {
    loading: 'Cargando el escaneo…',
    failed: 'No se pudo cargar el escaneo.',
    hint: 'Arrastra para girar · Rueda o pellizca para acercar',
    front: 'Frente', side: 'Lado', close: 'Primer plano',
    reveal: 'Revelar', points: 'Dispersar', light: 'Luz',
    change: 'Cómo cambiarlo',
    changeLine: 'La cámara, el título y el crédito están en data/scene.json. Pide a tu IA:',
    prompts: ['“Añade una vista desde arriba al visor.”', '“Usa mi escaneo assets/silla.spz, encuadrado de frente.”'],
  },
}

export const words = () => {
  const tag = (typeof navigator !== 'undefined' && navigator.language) || 'en'
  return WORDS[tag.toLowerCase().split('-')[0]] ?? WORDS.en
}

export const Page = () => {
  const w = words()
  return (
    <>
      <Viewer scene={scene} models={MODELS} words={w} />
      <div className="change">
        <style>{`
          .change { width: min(var(--kunobi-prose-width, 1200px), 100%); margin: 0 auto; box-sizing: border-box; padding: 0 24px 40px; font-family: Figtree, system-ui, sans-serif; }
          .change div { border: 1px solid hsl(var(--border)); background: hsl(var(--card)); border-radius: 20px; padding: 16px 20px; color: hsl(var(--muted-foreground)); font-size: 14px; line-height: 1.6; }
          .change b { color: hsl(var(--foreground)); font-weight: 600; margin-right: 8px; }
          .change span { display: inline-block; margin: 4px 8px 0 0; padding: 4px 10px; border-radius: 999px; background: hsl(var(--muted)); color: hsl(var(--foreground)); }
        `}</style>
        <div>
          <b>{w.change}</b>{w.changeLine}<br />
          {w.prompts.map((p) => <span key={p}>{p}</span>)}
        </div>
      </div>
    </>
  )
}
