export const Pagina = () => {
  const t = useI18n(WORDS, translations)
  const nombresLosetas = React.useMemo(
    () => Object.fromEntries(t.data('losetas', losetas).tipos.map((tipo) => [tipo.id, tipo.nombre])),
    [t],
  )
  const paleta = React.useMemo(() => t.data('jugadores', jugadores).colores, [t])
  return <Castillos w={t.w} nombresLosetas={nombresLosetas} paleta={paleta} />
}
