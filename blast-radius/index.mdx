export const pick = () => {
  const tag = (typeof navigator !== 'undefined' && navigator.language) || 'en'
  const lang = tag.toLowerCase().split('-')[0]
  return WORDS[lang] ? { w: WORDS[lang], locale: tag } : { w: WORDS.en, locale: 'en' }
}

export const Page = () => {
  const { w, locale } = pick()
  return <BlastRadius w={{ ...w, list: (xs) => list(xs, w.and) }} locale={locale} />
}
