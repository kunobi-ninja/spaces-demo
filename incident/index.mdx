export const span = (ms) => {
  const m = Math.max(0, Math.floor(ms / 60000))
  const h = Math.floor(m / 60)
  return { d: Math.floor(h / 24), h, m: m % 60 }
}

export const sentences = (w) => {
  const list = (xs) => (xs.length < 2 ? xs.join('') : w.story.and({ list: xs.slice(0, -1).join(', '), last: xs[xs.length - 1] }))
  return {
    duration: (ms) => { const s = span(ms); return s.d >= 2 ? w.durationDays({ count: s.d }) : s.h >= 1 ? w.durationHours({ hours: s.h, minutes: s.m }) : w.durationMinutes({ count: s.m }) },
    ago: (ms) => { const s = span(ms); return s.d >= 1 ? w.agoDays({ count: s.d }) : s.h >= 1 ? w.agoHours({ count: s.h }) : s.m >= 1 ? w.agoMinutes({ count: s.m }) : w.justNow },
    plain: (pm) => pm.status === 'resolved'
      ? w.plainResolved({ date: pm.date(pm.resolved), app: pm.app, days: pm.openDays })
      : w.plainOpen({ app: pm.app, date: pm.date(pm.started) }),
    summary: (pm) => {
      const p1 = pm.failed.length
        ? [w.story.failed({ release: pm.release, date: pm.date(pm.started), versions: list(pm.failed), lastGood: pm.lastGood }), pm.stalledAt ? w.story.stalled({ date: pm.date(pm.stalledAt) }) : ''].filter(Boolean).join(' ')
        : w.story.noFailed({ release: pm.release })
      const p2 = [pm.running ? w.story.running({ version: pm.running, ready: pm.podsReady, pods: pm.pods }) : '', pm.blocked.length ? w.story.delivery({ blocked: list(pm.blocked), gate: pm.gate }) : ''].filter(Boolean).join(' ')
      const lag = pm.detectLagDays != null ? w.story.unnoticed({ days: pm.detectLagDays }) : ''
      const state = pm.status === 'resolved' ? w.story.resolved({ date: pm.date(pm.resolved) }) : pm.status === 'mitigated' ? w.story.mitigated : pm.actions.length ? w.story.openChanged : w.story.openUnchanged
      return [p1, p2, [lag, state].filter(Boolean).join(' ')]
    },
  }
}

export const Page = () => {
  const t = useI18n(WORDS, translations)
  const w = React.useMemo(() => ({ ...t.w, ...sentences(t.w), locale: t.lang === 'en' ? 'en-GB' : t.tag }), [t])
  const sample = React.useMemo(() => t.data('cluster', snapshot), [t])
  const localize = React.useCallback((live) => t.data('timeline', live, start), [t])
  return <Incident snapshot={sample} start={start} localize={localize} w={w} />
}
